# Runtime target: Render free tier (512 MB RAM, ~0.1 shared vCPU), ONE uvicorn
# worker. The single-worker assumption is load-bearing for the cache design - see
# DEPLOYMENT.md before changing it.
#
# Every dependency here is loaded into a single 512 MB process, so unused
# packages cost real memory. Verify a package is actually imported before adding it.
#
# Versions below are the ones the test suite actually runs against. They were
# previously out of sync with the development venv on five exact pins, which meant
# local test results did not describe the deployed artifact.

# ── Database ───────────────────────────────────────────────────────────────
# psycopg 3, not SQLAlchemy. Every query in this codebase is hand-written SQL and
# there are no ORM models, so an ORM would cost ~30 MB RSS on a 512 MB box for
# machinery nothing uses. [binary] ships the libpq wheel so no system Postgres is
# needed to build; [pool] is the connection pool - connecting per request costs a
# 50-100 ms TLS handshake where the SQLite open it replaced was ~10 us.
psycopg[binary,pool]==3.3.4

# ID-token verification (shared/oidc.py). [crypto] pulls `cryptography`, which is
# required for RS256/ES256 - without it PyJWT silently supports only HMAC, which is
# exactly the algorithm this must never accept.
PyJWT[crypto]==2.10.1

# Loads backend/.env in local development. The platform sets real environment
# variables, so this is a convenience rather than a dependency of correctness.
python-dotenv==1.1.1

# ── Web framework ──────────────────────────────────────────────────────────
fastapi==0.140.1
uvicorn==0.51.0
python-multipart==0.0.32
pydantic==2.13.4

# ── Computation (mutual-funds domain) ──────────────────────────────────────
# NOTE: pandas opportunistically imports pyarrow when it is installed
# (pandas.compat.pyarrow), which costs ~45 MB RSS for no benefit here — we use
# no Arrow-backed dtypes, Parquet, or Feather. Do not add pyarrow back.
pandas==3.0.2
numpy==2.4.4
pyxirr==0.10.8

# ── Market data ────────────────────────────────────────────────────────────
# 1.5.2, not 1.3.0: yfinance does not cap curl_cffi, and 1.3.0 resolves against
# curl_cffi>=0.16, which it is incompatible with ("Fix yfinance breaking with
# curl_cffi>=0.16" is a 1.5.2 changelog entry). The pin was also drifting from what
# was actually installed, which is how the dividendYield scale change went unnoticed.
yfinance==1.5.2
nsepython==2.97
# Zerodha Kite Connect SDK for equity portfolio sync (OAuth + holdings + P&L)
kiteconnect==5.2.1
# nselib removed: its only import site was market_indices._fetch_nselib_series, a
# "tier 4" benchmark fallback that no code path ever called. That function is deleted,
# so the package is now a pure install-size and import-surface cost.
requests==2.33.1
# Not imported directly anywhere; it is a hard pandas dependency and installs
# regardless. Pinned only so the resolved set is reproducible.
python-dateutil==2.9.0.post0

# ── File parsing ───────────────────────────────────────────────────────────
# PINNED DELIBERATELY AT 0.9.1, NOT 1.3.0.
# This file previously pinned 1.3.0 while every local test ran against 0.8.1.
# Upgraded to 0.9.1 to resolve a strict pdfminer.six dependency conflict with pdfplumber.
# Upgrading to 1.3.0 is still a major-version jump on PDF parsing
# with no test coverage over real CAS documents.
casparser==0.9.1
pdfplumber==0.11.10
pypdf==6.14.2
# Excel engines for broker P&L uploads: openpyxl handles .xlsx, xlrd handles
# legacy .xls. Neither is imported directly — pandas selects one by file
# extension at runtime, so both must stay installed even with zero grep hits.
openpyxl==3.1.5
xlrd==2.0.2
# camelot-py pulls opencv-python-headless (~60 MB RSS) as a hard dependency.
# Both are deferred to first AIS upload via lazy import in ais_parser.py.
# Pinned because AIS table extraction accuracy is version-sensitive.
camelot-py==2.0.0

# ── Testing ────────────────────────────────────────────────────────────────
pytest>=9.0.0
pytest-cov>=6.0.0
httpx>=0.28.1
# Parses every SQL literal in the tree as Postgres without needing a server, which
# is what catches a typo or leftover SQLite syntax before it reaches a deploy.
sqlglot>=27.0.0
